type 'a list =
| []
| (::) of 'a * 'a list

[] (* La liste vide *)
1 :: [] (* La liste contenant uniquement 1 *)
1 :: (2 :: []) (* La liste contenant 1, 2 *)
        \end{minted}

        \small Structures récursives

        \pause

        \begin{minted}{ocaml}
let rec sum (l : int list) : int =
    match l with
    | [] -> 0
    | x :: q -> x + sum q

sum [] (* 0 *)
sum (1 :: []) (* 1 *)
sum (1 :: (2 :: [])) (* 3 *)
        \end{minted}

    match l with
    | [] -> 0
    | x :: q -> x + sum q
    \end{minted}

    \vspace{0.5cm}

    \begin{minted}{ocaml}
let sum (l : int list) : int =
    let res = ref 0 in
    let l_ref = ref l in
    while !l_ref <> [] do
        res := !res + List.hd !l_ref;
        l_ref := List.tl !l_ref
    done;
    !res

(* 499999500000 *)
sum (List.init 1_000_000 (fun i -> i))
    \end{minted}
\end{frame}
