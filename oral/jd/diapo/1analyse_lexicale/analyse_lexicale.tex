\begin{frame}
    \frametitle{Analyse lexicale}

    But~: construire, à partir du code source, une liste de lexèmes plus faciles
    à traiter que du texte brut.

    \pause

    \begin{definition}[lexème]
      Un lexème est un couple \mintinline{text}{(type, valeur)}. Le \textbf{type}
      doit figurer dans une liste que l'on a prédéfini, et la \textbf{valeur}
      doit correspondre au type tel qu'on l'a prédéfini.

      Les types de lexèmes sont également appelés \textbf{symboles terminaux}.
    \end{definition}

    \pause

    Exemples~: \mintinline{text}{(int, 12)},
    \mintinline{text}{(string, "Hello, world")}, \mintinline{text}{(equals, =)}.

\end{frame}
