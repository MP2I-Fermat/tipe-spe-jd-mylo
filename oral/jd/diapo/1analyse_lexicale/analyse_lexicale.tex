      Les types de lexèmes sont également appelés \textbf{symboles terminaux}.
    \end{definition}

    \pause

    Exemples~: \mintinline{text}{(int, 12)},
    \mintinline{text}{(string, "Hello, world")}, \mintinline{text}{(equals, =)}.
