\subsection{Algorithme LR(0)}

\begin{frame}
    \frametitle{Algorithme LR(0)}

    \begin{itemize}
        \item \textit{\textbf{L}eft to right}~;
        \item \textit{\textbf{R}ightmost derivation}~;
        \item \textit{\textbf{0} lookahead symbols}.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Automates}

    \pause
    L’algorithme LR(0) repose sur l’utilisation \textbf{d’automates}.

    \pause
    \begin{definition}
        \textbf{États LR(0)} = ensemble de règles de grammaires munies de
        positions.

        \textbf{Transitions} étiquetées par des symboles.
    \end{definition}

    \pause
    L’automate LR(0) correspondant à une grammaire peut être construit à partir
    de cette grammaire.
\end{frame}
