\begin{frame}
    \frametitle{Fonctions récursives et boucles}

    \begin{itemize}
        \item On a souvent besoin d’itérer des instructions sur des données,
            c’est-à-dire de parcourir une certaine séquence de données en les
            traitant une par une
        \pause
        \item Deux méthodes existent~: boucles et fonctions récursives
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Boucles}

    Avantages~:
    \pause
    \begin{itemize}
        \item Plus facile à compiler ou à interpréter~;
        \item Moins de problèmes mémoires.
    \end{itemize}

    \pause
    Exemple~:
    \inputminted{ocaml}{exemples/0intro/somme_liste_boucle.ml}
